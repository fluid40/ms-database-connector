323 def _convert_to_iso_timestamp(self, sme_value) -> str:
324 """Convert different timestamp formats to ISO 8601 format.
325
326 Args:
327 sme_value: The value to convert (can be string, datetime, or numeric timestamp).
328
329 Returns:
330 ISO 8601 formatted datetime string.
331 """
332 if isinstance(sme_value, str):
333 try:
334 dt = datetime.fromisoformat(sme_value.replace("Z", "+00:00"))
335 return dt.isoformat()
336 except (ValueError, AttributeError, OverflowError):
337 _logger.warning(f"Could not parse timestamp string: {sme_value}")
338 return sme_value
339 elif isinstance(sme_value, datetime):
340 return sme_value.isoformat()
341 elif isinstance(sme_value, (int, float)):
342 try:
343 dt = datetime.fromtimestamp(sme_value, tz=timezone.utc)
344 return dt.isoformat()
345 except (ValueError, OSError, OverflowError):
346 _logger.warning(f"Could not convert numeric timestamp: {sme_value}")
347 return str(sme_value)
348 else:
349 return str(sme_value)
