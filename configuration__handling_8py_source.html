103 def _get_aas_registry_config(self):
104 """Load and validate AAS registry server configuration.
105
106 Reads the first JSON configuration file found in the aas_registry directory
107 and parses it into a ServerConfiguration object.
108
109 Raises:
110 ConfigurationError: If configuration directory does not exist, no configuration
111 files are found, or the configuration file is invalid (validation error).
112 """
113 config_path = Path(f"{CONFIG_BASE_PATH}/aas_registry")
114
115 if not config_path.exists() or not config_path.is_dir():
116 _logger.error(
117 f"AAS registry configuration path '{config_path}' not found or inaccessible."
118 )
119 raise ConfigurationPathError(
120 f"AAS registry configuration path '{config_path}' not found."
121 )
122
123 json_files = list(config_path.rglob("*.json"))
124
125 if not json_files or len(json_files) == 0:
126 _logger.error(
127 f"No AAS registry configuration files found in folder '{config_path}'."
128 )
129 raise ConfigurationFileError(
130 f"No AAS registry configuration files found in folder '{config_path}'."
131 )
132
133 _logger.debug(
134 f"Found {len(json_files)} AAS registry configuration files in folder '{config_path}'."
135 )
136
137 if len(json_files) > 1:
138 _logger.warning(
139 f"Multiple AAS registry configuration files found. Using the first one: '{json_files[0]}'."
140 )
141
142 try:
143 self._aas_registry_configuration = ServerConfiguration.model_validate_json(
144 json_files[0].read_text()
145 )
146 except ValidationError as ve:
147 _logger.error(f"Invalid AAS registry connection file: {ve}")
148 raise ConfigurationValidationError(
149 "Invalid AAS registry connection file."
150 ) from ve

222
223 if not json_files or len(json_files) == 0:
224 _logger.info(
225 f"No AAS repository configuration files found in folder '{config_path}'."
226 )
227 return
