137 def _is_valid_measurement_name(self, measurement: str) -> bool:
138 """Validate measurement names to prevent high-cardinality series creation.
139
140 Enforces naming rules to prevent accidental dynamic measurement names that could
141 create excessive series in InfluxDB. Valid names must start with a letter and
142 contain only [A-Za-z0-9_.:-] characters, plus optional numeric index segments
143 like [0] for submodel-element paths.
144
145 Args:
146 measurement: The measurement name to validate.
147
148 Returns:
149 bool: True if the measurement name is valid, False otherwise.
150 """
151 if not isinstance(measurement, str):
152 logger.error("Measurement name must be a string.")
153 return False
154
155 name = measurement.strip()
156
157 if not name:
158 logger.error("Measurement name must be non-empty.")
159 return False
160
161 # Keep names simple and predictable and allow indexed path segments like Mapping[0].
162 if len(name) > 128 or not re.fullmatch(
163 r"[A-Za-z][A-Za-z0-9_.:-]*(?:\[\d+\][A-Za-z0-9_.:-]*)*", name
164 ):
165 logger.error(
166 "Measurement name '%s' is invalid. Use a stable identifier starting with a letter and only [A-Za-z0-9_.:-] plus optional [index] segments.",
167 measurement,
168 )
169 return False
170
171 # Common accidental dynamic-name patterns that create high cardinality.
172 if re.search(r"[0-9]{10,}", name):
173 logger.error(
174 "Measurement name '%s' appears dynamic (long numeric fragment).",
175 measurement,
176 )
177 return False
178
179 if re.search(
180 r"[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}",
181 name,
182 ):
183 logger.error(
184 "Measurement name '%s' appears dynamic (UUID-like pattern).",
185 measurement,
186 )
187 return False
188
189 if re.search(r"\d{4}-\d{2}-\d{2}", name):
190 logger.error(
191 "Measurement name '%s' appears dynamic (date-like pattern).",
192 measurement,
193 )
194 return False
195
196 return True

198 def write_data(self, fields: dict, measurement: str, tags: dict, time: str) -> bool:
199 """Write data point to InfluxDB.
200
201 Writes a time-series data point with fields and tags to the configured bucket.
202 The 'timestamp' field is required and must be in ISO 8601 format.
203
204 Args:
205 fields: Dictionary of field values. Datetime values will be converted to ISO 8601 strings.
206 measurement: The name of the measurement to write to.
207 tags: Dictionary of tag key-value pairs for indexing.
208 time: Timestamp for the data point in ISO 8601 format (e.g. 2023-03-15T12:34:56.789Z).
209
210 Returns:
211 bool: True if the data was written successfully, False otherwise.
212 """
213 logger.debug(
214 f"Writing data to InfluxDB measurement '{measurement}' with tags: {tags}"
215 )
216
217 if not self._is_valid_measurement_name(measurement):
218 return False
219
220 if fields is None or not isinstance(fields, dict):
221 logger.debug(f"'{measurement}': Data must be a non-empty dictionary.")
222 return False
223
224 point: Point = Point(measurement)
225 for tag_key, tag_value in tags.items():
226 point.tag(tag_key, tag_value)
227 for field_key, field_value in fields.items():
228 if isinstance(field_value, datetime):
229 point.field(field_key, field_value.isoformat())
230 else:
231 point.field(field_key, field_value)
232 point.time(time, write_precision="ms")
233
234 logger.info(
235 f"Writing data to InfluxDB:\nMeasurement: '{measurement}'\nTags:\n{json.dumps(tags, indent=4)}'\nValues:\n{json.dumps(fields, indent=4, default=str)}"
236 )
237
238 return self.write_point(point)

311def _establish_connection(client: InfluxV2Client) -> bool:
312 """Establish connection to InfluxDB with retry logic.
313
314 Attempts to connect to the InfluxDB server with exponential backoff,
315 retrying for up to the configured connection timeout duration.
316
317 Args:
318 client: The InfluxV2Client instance to connect.
319
320 Returns:
321 bool: True if connection is established, False if timeout is exceeded.
322 """
323 start_time = time.time()
324 logger.info(
325 f"Try to connect to Influx DB '{client.url}' for {client.connection_time_out} seconds"
326 )
327 counter: int = 0
328 while True:
329 try:
330 root = client.ping()
331 if root:
332 logger.info(f"Connected to Influx DB at '{client.url}' successfully.")
333 return True
334 except requests.exceptions.ConnectionError:
335 pass
336 if time.time() - start_time > client.connection_time_out:
337 logger.error(
338 f"Connection to Influx DB timed out after {client.connection_time_out} seconds."
339 )
340 return False
341
342 counter += 1
343 logger.warning(f"Retrying connection to Influx DB (attempt: {counter})")
344 time.sleep(5)
