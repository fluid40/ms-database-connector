19@pytest.fixture
20def build_worker(mocker):
21 """Create a configurable PollingWorker test context.
22
23 This fixture returns a builder function so each test can override only the
24 dependency that matters for the scenario under test.
25
26 Sentinel behavior:
27 - `...` means "use the fixture default".
28 - `None` means "explicitly missing/unavailable" (used for negative paths).
29
30 The returned object is a ``SimpleNamespace`` with the worker and all
31 relevant mocks/providers for assertions.
32 """
33
34 def _build(
35 *,
36 polling_interval: int = 10,
37 target_references=...,
38 influx_client=...,
39 db_mapping=...,
40 mapper_instance=...,
41 ):
42 """Build one PollingWorker instance and its runtime dependency graph.
43
44 Args:
45 polling_interval: Polling interval forwarded to PollingWorker.
46 target_references: Value for ``app.state.target_references``.
47 Use ``...`` to leave attribute unset, which exercises the
48 "attribute missing" code path.
49 influx_client: Value returned by ``influx_client_provider``.
50 ``...`` creates a mock client, ``None`` simulates unavailable
51 client.
52 db_mapping: Value assigned to ``mapping_handler.db_mapping``.
53 ``...`` uses a minimal valid mapping, ``None`` simulates missing
54 mapping configuration.
55 mapper_instance: Mapper returned by ``mapper_factory``.
56 ``...`` creates a mock mapper that returns an empty batch.
57
58 Returns:
59 SimpleNamespace: Aggregated context with ``worker``, ``app``,
60 providers/factories, and created mocks to keep assertions concise.
61 """
62 app = FastAPI()
63
64 # Keep attribute absent when target_references is the sentinel; several
65 # tests validate the fallback behavior in that exact state.
66 if target_references is not ...:
67 app.state.target_references = target_references
68
69 # Server handler is passed into PollingWorker and into mapper_factory.
70 server_handler = mocker.Mock()
71
72 # Default client is a strict interface mock; None can be injected to
73 # trigger "client unavailable" behavior.
74 if influx_client is ...:
75 influx_client = mocker.Mock(spec=IInfluxClient)
76
77 mapping_handler = mocker.Mock()
78 # Use a minimal mapping by default so collection can proceed unless a
79 # test intentionally overrides this input.
80 mapping_handler.db_mapping = (
81 {"measurement": {"field": "path"}} if db_mapping is ... else db_mapping
82 )
83
84 if mapper_instance is ...:
85 mapper_instance = mocker.Mock()
86 # Empty batch is the safest default for generic setup paths.
87 mapper_instance.map_smes_to_influx.return_value = {}
88
89 # Providers/factories are mocks so tests can assert interaction details.
90 influx_client_provider = mocker.Mock(return_value=influx_client)
91 mapping_handler_provider = mocker.Mock(return_value=mapping_handler)
92 mapper_factory = mocker.Mock(return_value=mapper_instance)
93
94 runtime_deps = AppRuntimeDeps(
95 config_provider=mocker.Mock(),
96 mapping_handler_provider=mapping_handler_provider,
97 influx_client_provider=influx_client_provider,
98 server_handler_factory=mocker.Mock(),
99 config_loader_factory=mocker.Mock(),
100 mapper_factory=mapper_factory,
101 )
102
103 worker = PollingWorker(
104 app,
105 server_handler,
106 polling_interval=polling_interval,
107 runtime_deps=runtime_deps,
108 )
109
110 # Expose all frequently asserted components through one object.
111 return SimpleNamespace(
112 app=app,
113 worker=worker,
114 server_handler=server_handler,
115 runtime_deps=runtime_deps,
116 influx_client=influx_client,
117 mapping_handler=mapping_handler,
118 mapper_instance=mapper_instance,
119 influx_client_provider=influx_client_provider,
120 mapping_handler_provider=mapping_handler_provider,
121 mapper_factory=mapper_factory,
122 )
123
124 return _build
