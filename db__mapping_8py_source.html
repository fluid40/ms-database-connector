6class MappingTargetType(str, Enum):

9 Attributes:
10 FIELD: Target is an InfluxDB field (numeric or string values).
11 TAG: Target is an InfluxDB tag (indexed string values).
12 TIMESTAMP: Target is the timestamp (exactly one per measurement).

20class MeasurementMapping(RootModel[dict[str, MappingTargetType]]):
