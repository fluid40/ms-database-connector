23@dataclass(frozen=True)
24class AppRuntimeDeps:
25 """Composition object for runtime providers and factories.
26
27 Keeping runtime wiring in one immutable object simplifies testing by reducing
28 monkeypatch targets from many globals to a single dependency container.
29 """

31 config_provider: Callable[[], ServiceConfiguration]

39def build_app_runtime_deps() -> AppRuntimeDeps:

51@lru_cache(maxsize=1)
52def get_service_configuration() -> ServiceConfiguration:
53 """Load the runtime service configuration from environment.
54
55 Uses a singleton cache to avoid repeated file parsing. The configuration file
56 path is read from the DBC_CONFIGURATION_FILE environment variable.
57
58 Returns:
59 ServiceConfiguration: The loaded and parsed service configuration.
60
61 Raises:
62 Exception: If configuration file cannot be loaded or parsed.
63 """
64 configuration_file = os.getenv("DBC_CONFIGURATION_FILE", "")
65 return load_configuration(configuration_file)

68@lru_cache(maxsize=1)
69def get_influx_client() -> IInfluxClient | None:
70 """Create and cache an InfluxDB client using current service configuration.
71
72 Initializes a singleton InfluxDB client based on the active service configuration.
73 The client is cached to avoid repeated initialization.
74
75 Returns:
76 IInfluxClient | None: The initialized InfluxDB client instance, or None if
77 initialization fails (e.g., missing credentials or invalid configuration).
78 """
79 configuration = get_service_configuration()
80 return initialize_db_connection(configuration)

83def reconnect_influx_client() -> IInfluxClient | None:
84 """Force a fresh InfluxDB client initialization.
85
86 Clears the cached client and initializes a new one. This is typically called
87 when the user requests to reconnect to the InfluxDB server via the API.
88
89 Returns:
90 IInfluxClient | None: The newly initialized InfluxDB client instance,
91 or None if initialization fails.
92 """
93 get_influx_client.cache_clear()
