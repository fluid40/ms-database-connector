50def _track_startup(state: dict, key: str, flag: str):
51 """Context manager helper for tracking initialization steps.
52
53 Args:
54 state: The startup state dictionary to track initialization steps.
55 key: The key representing the initialization step.
56 flag: The flag to set in the startup state upon successful completion.
57
58 Returns:
59 StartupTracker: A context manager for tracking the initialization step.
60 """
61
62 class StartupTracker:
63 def __init__(self, state, key, flag):
64 self.state = state
65 self.key = key
66 self.flag = flag or key
67
68 def __enter__(self):
69 return self
70
71 def __exit__(self, exc_type, exc_val, exc_tb):
72 if exc_type is None:
73 self.state[self.flag] = True
74 else:
75 self.state["errors"].append(f"{self.key}: {exc_val}")
76 return False
77
78 return StartupTracker(state, key, flag)

203class PollingWorker:
204 """Background worker that polls AAS submodel elements and writes values to InfluxDB.
205
206 Uses the polling_interval from the service configuration to schedule
207 periodic data collection. The worker holds references to the FastAPI app,
208 AAS server handler, and the InfluxDB client established during startup.
209
210 Args:
211 app: The FastAPI application instance (holds target references in state).
212 server_handler: The connected AAS server handler.
213 polling_interval: Interval in seconds between polling cycles.
214 """
215
216 def __init__(
217 self,
218 app: FastAPI,
219 server_handler: ServerHandler,
220 polling_interval: int,
221 runtime_deps: AppRuntimeDeps,
222 ) -> None:
223 self._app = app
224 self._server_handler = server_handler
225 self._polling_interval = polling_interval
226 self._runtime_deps = runtime_deps
227 self._task: asyncio.Task | None = None
228 self._stop_event = asyncio.Event()
229
230 def start(self) -> None:
231 """Schedule the polling loop as an asyncio background task."""
232 self._stop_event.clear()
233 self._task = asyncio.create_task(self._run(), name="polling-worker")
234 _logger.info("Polling worker started (interval=%ds).", self._polling_interval)
235
236 async def stop(self) -> None:
237 """Signal the polling loop to stop and await its completion."""
238 _logger.info("Stopping polling worker.")
239 self._stop_event.set()
240 if self._task is not None:
241 await asyncio.wait([self._task], timeout=self._polling_interval + 5)
242 self._task = None
243
244 async def _run(self) -> None:
245 """Main polling loop.
246
247 On each cycle the worker uses the AAS server handler and InfluxDB
248 client established during startup. AAS and AIMC metadata are preloaded
249 once at startup via _preload_aas_metadata and are not re-fetched here.
250 """
251 while not self._stop_event.is_set():
252 try:
253 await self._poll_once()
254 except Exception:
255 _logger.exception("Error during polling cycle.")
256 try:
257 await asyncio.wait_for(
258 asyncio.shield(asyncio.ensure_future(self._stop_event.wait())),
259 timeout=self._polling_interval,
260 )
261 except asyncio.TimeoutError:
262 pass # normal – interval elapsed, continue loop

287 def collect_influx_points(
288 self,
289 ) -> PollingCyclePayload | None:
290 """Collect mapped InfluxDB points required for one polling cycle.
291
292 Reads current runtime dependencies and target references, then maps
293 SMEs to InfluxDB points that are ready for persistence.
294
295 Returns:
296 PollingCyclePayload | None: Collection result containing the
297 initialized InfluxDB client and mapped points, or None if
298 required prerequisites are unavailable or mapping fails.
299 """
300
301 influx_client = self._runtime_deps.influx_client_provider()
302 if influx_client is None:
303 _logger.warning("InfluxDB client unavailable – skipping cycle.")
304 return None
305
306 target_references = getattr(self._app.state, "target_references", [])
307 if not target_references:
308 _logger.warning("No target references available – skipping cycle.")
309 return None
310
311 db_mapping = self._runtime_deps.mapping_handler_provider().db_mapping
312 if db_mapping is None:
313 _logger.warning("No DB mapping configuration available – skipping cycle.")
314 return None
315
316 mapper = self._runtime_deps.mapper_factory(
317 server_handler=self._server_handler,
318 db_mapping=db_mapping,
319 target_references=target_references,
320 )
321
322 try:
323 influx_points = mapper.map_smes_to_influx()
324 except Exception:
325 _logger.exception("Failed to map SMEs to InfluxDB points.")
326 return None
327
328 return PollingCyclePayload(
329 influx_client=influx_client,
330 influx_points=influx_points,
331 )

360async def lifespan(app: FastAPI):
361 """Lifespan function for the FastAPI application.
362
363 Handles startup initialization and shutdown cleanup events. Initializes all
364 required services and validates connectivity before the application starts
365 serving requests.
366
367 Args:
368 app: The FastAPI application instance.
369
370 Yields:
371 None: Context is yielded after startup, allowing the app to run.
372
373 Raises:
374 Exception: If any required initialization step fails.
375 """
376 _logger.info("Starting up the microservice database connector.")
377 app.state.startup = _init_startup_state()
378 app.state.runtime_deps = build_app_runtime_deps()
379 runtime_deps: AppRuntimeDeps = app.state.runtime_deps
380
381 try:
382 _setup_service_config(app.state.startup, runtime_deps)
383 _setup_mapping_service(app.state.startup, runtime_deps)
384 _setup_influx_connection(app.state.startup, runtime_deps)
385 server_handler = _setup_aas_connection(app.state.startup, runtime_deps)
386 _preload_aas_metadata(app, server_handler, app.state.startup, runtime_deps)
387 except Exception:
388 _logger.exception("Startup initialization failed.")
389 raise
390
391 polling_interval = runtime_deps.config_provider().polling_interval
392 worker = PollingWorker(app, server_handler, polling_interval, runtime_deps)
393 worker.start()
394
395 yield
396
397 await worker.stop()
398 _logger.info("Shutting down the microservice database connector.")
