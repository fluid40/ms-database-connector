60 def reload_db_mapping_from_file(self) -> bool:
61 """Reload db mapping from disk.
62
63 Reads db_mapping.json and validates it to match the RawDbMapping model. Invalid or missing files
64 are logged but do not raise exceptions, allowing recovery via API.
65
66 Returns:
67 bool: True if a fully-validated mapping was loaded, False if file is
68 missing, invalid, or contains unfilled entries.
69 """
70 if not self._db_mapping_file.exists() or not self._db_mapping_file.is_file():
71 _logger.error(
72 "DB mapping file '%s' not found or inaccessible.",
73 self._db_mapping_file,
74 )
75 self._db_mapping = None
76 self._raw_db_mapping = None
77 return False
78
79 try:
80 raw_mapping_json = json.loads(
81 self._db_mapping_file.read_text(encoding="utf-8")
82 )
83 self._raw_db_mapping = RawDbMapping.model_validate(raw_mapping_json)
84
85 except json.JSONDecodeError as exc:
86 _logger.error(
87 "Invalid JSON in DB mapping file '%s': %s",
88 self._db_mapping_file,
89 exc,
90 )
91 self._db_mapping = None
92 self._raw_db_mapping = None
93 return False
94 except OSError as exc:
95 _logger.error(
96 "Failed to read DB mapping file '%s': %s",
97 self._db_mapping_file,
98 exc,
99 )
100 self._db_mapping = None
101 self._raw_db_mapping = None
102 return False
103 except ValidationError as exc:
104 _logger.error(
105 "Validation failed for DB mapping file '%s': %s",
106 self._db_mapping_file,
107 exc,

115 def update_db_mapping(self, db_mapping: DbMapping) -> bool:
116 """Store and persist a validated db mapping.
117
118 Serializes the typed mapping to raw model format and persists to disk.
119
120 Args:
121 db_mapping: A validated DbMapping object.
122
123 Returns:
124 bool: True if mapping was updated and persisted successfully,
125 False if persistence fails.
126 """
127 serialized_mapping: dict[str, dict[str, str | None]] = {
128 measurement_name: {
129 sink_path: target_type.value
130 for sink_path, target_type in measurement_mapping.root.items()
131 }
132 for measurement_name, measurement_mapping in db_mapping.root.items()
133 }
134
135 # Convert to RawDbMapping model
136 raw_db_mapping = RawDbMapping.model_validate(serialized_mapping)
137
138 self._db_mapping = db_mapping
139 self._raw_db_mapping = raw_db_mapping
140
141 try:
142 self._persist_db_mapping(raw_db_mapping)
143 except OSError as exc:
144 _logger.error(
145 "Failed to persist DB mapping to '%s': %s",
146 self._db_mapping_file,
147 exc,
148 )

154 self,
155 persist: bool = True,
156 ) -> bool:
157 """Validate if raw DB mapping is complete and update typed mapping.
158
159 Args:
160 persist: If True, persist the mapping to disk (default: True).
161
162 Returns:
163 bool: True if valid and stored/persisted successfully, False otherwise.
164 """
165 if self._raw_db_mapping is None:
166 _logger.error("No raw DB mapping available to update from.")
167 self._db_mapping = None
168 return False
169
170 if self._raw_db_mapping.has_unfilled_entries():
171 _logger.info(
172 "DB mapping contains unfilled entries (null values). "
173 "Keeping raw template and clearing typed mapping."
174 )
175 self._db_mapping = None
176 return False
177
178 try:
179 db_mapping = DbMapping.model_validate(self._raw_db_mapping.model_dump())
180 except ValidationError as exc:
181 _logger.error("DB mapping validation failed: %s", exc)
182 self._db_mapping = None
183 return False
184
185 self._db_mapping = db_mapping
186
187 if not persist:
188 return True

191 self._persist_db_mapping(self._raw_db_mapping)
192 except OSError as exc:

196 exc,
